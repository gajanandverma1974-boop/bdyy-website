    <script>
        let currentStep = 1;
        const totalSteps = 3;

        /* --- Core Step-by-Step Navigation --- */
        function nextStep() {
            if (currentStep < totalSteps) {
                // Hide current step
                document.getElementById(`step-${currentStep}`).classList.remove('active');
                
                // Advance step
                currentStep++;
                
                // Show next step
                document.getElementById(`step-${currentStep}`).classList.add('active');
            }
        }

        /* --- Finalization and Celebration Effect --- */
        function finalizeGreeting() {
            // Hide the main card
            document.getElementById('greeting-container').classList.add('final-state');
            
            // Show the final message
            const finalMessage = document.getElementById('final-message');
            finalMessage.classList.add('show');
            
            // Trigger the fireworks/particle burst effect
            createFireworks(15); // Create 15 bursts
        }

        /* --- Background Particle Generator JS (Aesthetic Effect) --- */
        const particleContainer = document.getElementById('particle-container');
        const particleCount = 50;

        for (let i = 0; i < particleCount; i++) {
            const particle = document.createElement('div');
            particle.classList.add('particle');
            
            // Random position and delay
            const x = Math.random() * 100;
            const y = Math.random() * 100;
            const delay = Math.random() * 10;
            
            // Custom CSS variables for the animation (random float direction)
            particle.style.left = `${x}vw`;
            particle.style.top = `${y}vh`;
            particle.style.animationDelay = `${delay}s`;
            particle.style.setProperty('--dx', `${Math.random() * 40 - 20}px`); // -20px to 20px
            particle.style.setProperty('--dy', `${Math.random() * 40 - 20}px`);
            
            particleContainer.appendChild(particle);
        }

        /* --- Fireworks Burst Effect (Simplified CSS/JS) --- */
        function createFireworks(count) {
            const finalMessage = document.getElementById('final-message');
            for (let i = 0; i < count; i++) {
                // Random position for the burst
                const posX = Math.random() * window.innerWidth;
                const posY = Math.random() * window.innerHeight;

                const burstSize = 10 + Math.random() * 10; // 10 to 20 stars per burst

                for (let j = 0; j < burstSize; j++) {
                    const star = document.createElement('div');
                    star.classList.add('firework-star');
                    
                    // Initial center position
                    star.style.left = `${posX}px`;
                    star.style.top = `${posY}px`;
                    star.style.animationDelay = `${i * 0.1}s`; // Stagger the bursts
                    
                    // Small random movement for a realistic scatter
                    const deltaX = (Math.random() - 0.5) * 150;
                    const deltaY = (Math.random() - 0.5) * 150;

                    // This is a simple explosion effect - real fireworks would be complex
                    star.style.setProperty('animation-name', 'firework-burst');
                    star.addEventListener('animationend', () => star.remove()); // Clean up after burst

                    finalMessage.appendChild(star);
                }
            }
        }

    </script>
